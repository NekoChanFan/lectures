\section{Конечные преобразователи}
\dfn{Конечный преобразователь}{Шестёрка $S=(Q,\Sigma,\Delta,q_0, \delta,F)$, где:\\
Q --- множество состояний,\\
$\Sigma$ --- входной алфавит,\\
$\delta\subset Q\times\Sigma^*\times\delta \times Q$ --- конечное множество переходов(программа),\\
$q_0$ --- начальное состояние,\\
$F \subseteq Q$ --- множество закл. состояний}

\dfn{Конфигурация}{
  Тройка $(q,u,v)$, где $q\in Q, u\in \Sigma^*, v\in \Delta^*$
}
\dfn{Переход за один шаг($\vdash$)}{
  $(q,u,v)\vdash_S(p,x,y), \text{ если существует }w\in\Sigma^*,t\in\Delta^*:\\
  u=wx,y=vt$ и $q,w\rightarrow p,v \in \delta$
}
\dfn{Отношение, вычисляемое S}{
  $R(S)= \{(x,y):(q_0,x,\varepsilon) \vdash^*_S (q_f,\varepsilon,y)\text{ для некоторого }q_f\in F\}$
}
\dfn{Детерминированный КП}{
Шестёрка $S=(Q,\Sigma,\Delta,\delta,q_0,F),\text{ где }\delta: Q\times\Sigma\rightarrow Q\times\Delta^*$}

\section{Морфологический анализ}
\noindent Задачи морфологического анализа:
\begin{enumerate}
  \item Дано слово, поставить его в нужную форму.
  \item Дано слово, найти его начальную форму.
\end{enumerate}

\mprop{Рассматриваемые характеристики}{
  \begin{itemize}
    \item N --- существительное(noun)
    \item V --- глагол(verb)
    \item SG --- единственное число(singular)
    \item PL --- множественное число(plural)
  \end{itemize}
}
\ex{}{
  cat+N+PL $\Rightarrow$ cats\\
  cats $\Rightarrow$ cat+N+PL
}

\dfn{Морфологические правила}{
  Определяют, как образуются формы.\\
  Пример: английский язык, множественное число = ед. число + s.
}

\dfn{Орфографические правила}{
  Определяют, как меняется написание.\\
  Пример --- англ. язык, мн. число:\\
  Слово заканчивается на s $\rightarrow$ вставить е.\\
  Слово заканчивается на y $\rightarrow$ заменить на iе.\\
  Слово является исключением $\rightarrow$ особый случай.
}

\dfn{Основа}{<<Главная часть>> слова.}

\dfn{Аффексы}{Части слова, приписываемые к основе:
\begin{enumerate}
  \item Префиксы: пишутся спереди.
  \item Суффиксы: пишутся сзади.
  \item Инфиксы: пишутся внутри.
  \item Циркумфиксы: спереди и сзади.
\end{enumerate}}

\nt{Для морфологического анализатора необходимы:
\begin{enumerate}
  \item Словарь --- список основ и аффексов.
  \item Морфотактика --- правила объединения морфем.
  \item Орфографические правила.
\end{enumerate}}

\dfn{Обратное отношение}{
  $S=(Q,\Sigma,\Delta,\delta,q_0,F), R(S)\subseteq\Sigma^*\times\Delta$ --- конечный преобразователь.\\
  $S'=(Q,\Delta,\Sigma, \delta',q_0,F)$ --- конечный преобразователь, для которого множества входных и выходных данных поменяны местами.\\
  А программа имеет вид:\\
  $q,x\rightarrow p,y\in\delta\Leftrightarrow q,y\rightarrow p,x\in\delta'$\\
  Тогда:\\
  $R(S')=(R(S))^{-1}$\\
  $(x,y)\in R(S')\Leftrightarrow(y,x)\in R(S)$
}

\dfn{Выравнивание строк}{
  $S[1\dots n],T[1\dots m]\in\Sigma\\
  \text{---} \notin\Sigma$\\
  $\sigma$ --- стоимость однобуквенного выравнивания.\\
  $\sigma(a,a)\geq0\\
  \sigma(\text{---},a)\leq0\\
  \sigma(a,\text{---})\leq0\\
  \sigma(\text{---},\text{---})\leq0$

  Выравнивание строк: $S'[1\dots l],T'[1\dots l]$ такие, что $S,T$ --- проекции $S',T'$ на $\Sigma$
}
\dfn{Стоимость выравнивания $S',T'$}{
  $\sigma(S',T')=\displaystyle\sum_{i=1}\sigma(S'[i],T'[i])$
}
\dfn{Оптимальное выравнивание}{
  Выравнивание наибольшей стоимости.
}
\dfn{Редакционное расстояние}{
  Наименьшее число операций вставки символа, удаления символа и замены символа для преобразования $S$ в $T$.
}

\dfn{Алгоритм Вагнера-Фишера}{
  $V(i,j)$ --- наибольшая стоимость выравнивания $S[1\dots i]$ и $T[1\dots j]$\\
  $V(0,0)=0$\\
  $V(i+1,0)=V(i,0)+\sigma(S[i+1],\text{---})$\\
  $V(0,j+1)=V(0,j)+\sigma(\text{---},T[j+1])$\\
  $V(i+1,j+1)=max\{V(i,j)+\sigma(S[i+1],T[j+1]),\\
    V(i+1,j)+\sigma(\text{---},T[j+1]),\\
    V(i,j+1)+\sigma(S[i+1],\text{---}),
  \}$\\
Время: O(mn)\\
Память: O(mn), если нужно само выравнивание\\
O(min\{m,n\}), если нужна только стоимость
}

\mprop{Версия алгоритма с хорошей памятью}{
  $W(i,j)$ --- наибольшая стоимость выравнивания $S[n-i+1\dots n]$ и $T[m-j+1\dots m]$

  $m\leq n$

    \begin{itemize}
      \item 
    \end{itemize}
  \item  Пусть $m>1$.
  \item  Пусть $T_1=T[1\dots\frac{m}{2}],T_2=T[\frac{m}{2}+1\dots m]$.
  \item  Вычислять последние строки таблицы V для $S$ и $T_1$, таблицы $W$ для $S$ и $T_2$

\section{Формула Байеса}
\dfn{Условная вероятность}{
  $P(A\backslash B)=\dfrac{P(AB)}{P(B)}$
}
\dfn{Полная группа событий}{
  $\Omega=H_1\cup H_2\cup\dots\cup H_m$,$H_i\cap H_j=\emptyset$
}
\dfn{Независимые события}{
  $P(A\backslash B)=P(A)$ или $P(AB)=P(A)P(B)$
}
\dfn{Формула полной вероятности}{
  $P(A)=\displaystyle\sum_{i=1}P(A\backslash H_i)P(H_i)$
}
\dfn{Формула Байеса}{
  $P(A\backslash B)=\dfrac{P(B\backslash A)P(A)}{P(B)}$\\
  $P(H_i\backslash A)=\dfrac{P(A\backslash H_i)P(H_i)}{P(A)}=\dfrac{P(A\backslash H_i)P(H_i)}{\displaystyle\sum_{i=1}^nP(A\backslash H_i)P(H_i)}$
}
\mprop{Применение формулы Байеса}{
  $c$ --- правильное написание слова.
  $t$ --- слово с ошибкой.
  $P(c\backslash t)$ --- вероятность того, что вместо $c$ написать $t$\\
  $P(c\backslash t)=\dfrac{P(t\backslash c)P(c)}{P(t)}$\\
  $P(c)$---вероятность употребления слова $c$\\
  $P(t)$---вероятность написать $t$\\
  $P(t\backslash c)$ --- вероятность допустить ошибку, которая превращает $c$ в $t$
}
\dfn{Корпус}{
  Набор текстов(возможно размеченных)
}
\nt{
  $P(c)$ приблизительно равно частоте слова в корпусе\\
  $P(t\backslash c)$
}
\ex{Типичные ошибки}{
  \begin{enumerate}
    \item Вставка символа: количество $\Rightarrow$ кол\underline{л}ичество
    \item Удаление: солнце $\Rightarrow$ сонце
    \item Замена: единица $\Rightarrow$ еденица
    \item Перестановка соседних символов: конечный $\Rightarrow$ конченый
  \end{enumerate}
}
\mprop{Определим некоторые величины}{
  \begin{itemize}
    \item $del[x,y]$ --- число раз, когда вместо xy было написано x
    \item $ins[x,y]$ --- число раз, когда вместо x было написано xy
    \item $sub[x,y]$ --- число раз, когда вместо x было написано y
    \item $trans[x,y]$ --- число раз, когда вместо xy было написано yx
    \item $count[\alpha]$ --- число раз, когда написано $\alpha$
  \end{itemize}
}
$P(t\backslash c)=\begin{cases}
  \dfrac{del[c_{p-1},c_p]}{count[c_{p-1}c_{p}]} \text{ для удаления},\\
  \dfrac{ins[c_{p-1},t_p]}{count[c_{p-1}]} \text{ для вставки},\\
  \dfrac{sub[c_p,t_p]}{count[c_p]} \text{ для замены},\\
  \dfrac{trans[c_p,c_{p+1}]}{count[c_pc_{p+1}]} \text{ для перестановки}
\end{cases}$
\dfn{Матрица ошибок}{
  Матрица значений del,ins,sub,trans,count.
}

\nt{
  Возвращаясь к фармуле Байеса для поиска вероятности ошибок, в формуле\\
  $P(c\backslash t)=\dfrac{P(t\backslash c)P(c)}{P(t)}$\\
  Мы научились вычислять $P(c),P(t\backslash c)$.\\
  А вычислять $P(t)$ не нужно, так как\\
  $\hat{c}=argmax P(c\backslash t)=argmax \dfrac{P(t\backslash c)P(c)}{P(t)} = argmax P(t\backslash c)P(c)$
}

\section{N-граммы}
\dfn{N-грамма}{
  N-грамма --- последовательность из N слов.\\
  N=1 --- униграмма\\
  N=2 --- биграмма\\
  N=3 --- триграмма
}

\dfn{Скалярное произведение}{
  $x=\mvec{x}{n}$\\
  $y=\mvec{y}{n}$\\
  Скалярное произведение: $\displaystyle\sum_{i=1}^{n}x_iy_i=(x,y)$
}
\ex{}{
  $T_1,T_2$ --- два текста\\
  $n$ --- число слов в текстах\\
  $x_i$ --- число вхождений $i$-го слова в $T_1$\\
  $y_i$ --- то же, что $x_i$, но для $T_2$\\
  $(x,y)$ --- косинусное расстояние между $T_1$ и $T_2$.\\
  Косинусное потому что $cos(\phi)=\dfrac{(x,y)}{|x|\cdot|y|}$

  Идея состоит в том, что если $T_1$ и $T_2$ "близки", то $(x,y)$ --- большое. А если $T_1$ и $T_2$ не связаны тематически, то $(x,y)$ близко к 0.

  Возникают вопросы, связанные с тем, что слова имеют много форм:
  \begin{enumerate}
    \item Можно убрать из $T_1,T_2$ артикли, предлоги, союзы и т.п.
    \item Нужно ли различать формы одного слова.
  \end{enumerate}
}

\mprop{Подсчёт вероятностей последовательностей слов}{
  $P(AB)=P(A|B)P(B)$\\
  $P(ABC)=P(A|BC)P(B|C)P(C)$\\
  $\mvec{w}{n}$ --- последовательность слов(текст)\\
  $P\mvec{w}{n}$ --- вероятность $\mvec{w}{n}$\\
  $P\mvec{w}{n}=P(w_1)P(w_2|w_1)P(w_3|w_2w_1)\dots P(w_n|w_1\dots w_{n-1})$

  В данной ситуации помогут N-граммы.\\
  Предположим, что слово $w_i$ зависит только от N-1 предыдущего слова.\\
  $P(w_i|w_1,\dots,w_{i-1})=P(w_i|w_{i-N+1}\dots w_{i-1})$
}
\ex{Частные случаи N-грамм}{
  \begin{enumerate}
    \item N=1.\\
      $P(w_i|w_1\dots w_{i-1})=P(w_i)$
    \item N=2.\\
      $P(w_i|w_1\dots w_{i-1})=P(w_i|w_{i-1})$
    \item N=3.\\
      $P(w_i|w_1\dots w_{i-1})=P(w_i|w_{i-2}w_{i-1})$
  \end{enumerate}
}

\nt{
  Если в языке V слов, то число N-грамм составляет V$^N$.
  Тогда возникает проблема:
  \begin{itemize}
    \item Если N "маленькое", то не учитываются "длинные" контексты.
    \item Если N "большое", то $V^N$ больше размера корпуса.
  \end{itemize}
  При "больших" N получается переобучение.\\
  $P\mvec{w}{n}=\prod_{i=1}^nP(w_i|w_{i-N+1}\dots w_{i-1})$\\
  $w_{N-2}=w_{N-1}=\dots=w_0=<s>$ --- спец слово\\
  При $N=2. P\mvec{w}{n}=P(w_1)p(w_2|w_1)P(w_3|w_2)\dots P(w_n|w_{n-1})$
}

\section{Сглаживание}
\dfn{Сглаживание}{
  При большом N часто $c(w_{i-1}w_i)=0, P(w_i|w_{i-1})=0.$\\
  0 могут быть в знаменателе в формулах, указанных выше.

  Сглаживание --- изменение вероятностей так, чтобы:
  \begin{enumerate}
    \item не осталось нулей;
    \item вероятности изменились "не очень сильно".
  \end{enumerate}
}

\nt{
  Пусть есть V наблюдаемых объектов (N-граммы).\\
  Проведено K наблюдений: $\{a_1,\dots a_V\}$.\\
$c_i$ --- число наблюдений $a_i$\\
$p_i$ --- вероятность $a_i$\\
$p_i=\frac{c_i}{k}$}

\mprop{Аддитивное сглаживание(Сглаживание Лапласа)}{
      Считаем, что каждый объект наболюдается на один раз чаще, чем на самом деле.\\
      $p_i^*$ --- сглаживание вероятностей.\\
      $p_i^*=\dfrac{c_i+1}{K+V}$\\
      $K+V$ --- число наблюдений с учётом добавленных\\
      $\displaystyle\sum_{i=1}^{V}p_i^*=\displaystyle\sum_{i=1}^{V}\dfrac{c_i+1}{K+V}=\dfrac{1}{K+V}\displaystyle\sum_{i=1}^{V}c_i+\displaystyle\sum_{i=1}^{V}\dfrac{1}{K+V}=$\\
      $=\dfrac{K}{K+V}+\dfrac{V}{K+V}=1$\\
      получается, что $p_i^*\geq0$\\
      $c_i^*$ --- сглаживание количества.
      $c_i^*=p_i^*K=(c_i+1)\dfrac{K}{K+X}$\\
      $\displaystyle\sum_{i=1}^{V}c_i^*=\displaystyle\sum_{i=1}^{V}p_i^*K=K$
}

\mprop{Сглаживание Уиттена-Белла}{
  $V,K,c_i$ --- означают то же, что и в предыдущем алгоритме.\\
  Суть алгоритма:\\
  Оценить вероятность того, что на (K+1)-м шаге появится $a_i$.\\
  Вероятность нового $a_i\approx$ вероятность появления события впервые.\\
  Введём величину:\\
  $Z=\displaystyle\sum_{c_i=0}1$ --- количество событий, которые никогда не наблюдались.
  Тогда дадим всем им такую вероятность:\\
  $\displaystyle\sum_{c_i=0}p_i^*=\dfrac{T}{K+T}$, где $T$ --- количество событий, которые произошли хотя бы один раз.\\
  $\dfrac{T}{K+T}$ --- вероятность увидеть событие впервые.

  Тогда вероятности определяются следующи образом:\\
  $p_i^*=\dfrac{T}{Z(K+T)}$, если $c_i=0$\\
  $p_i^*=\dfrac{c_i}{K+T}$, если $c_i>0$

  Количества вычисляются следующим образом:\\
  $c_i^*=p_i^*K$\\
  $c_i^*=\dfrac{T}{Z}\cdot\dfrac{K}{K+T}$, если $c_i=0$\\
  $c_i^*=c_i\cdot\dfrac{K}{K+T}$, если $c_i>0$
}

\dfn{КЗ-грамматики}{
  $G=(N,\Sigma,P,S)$\\
  Правила имеют вид: $\xi A\eta\rightarrow \xi\alpha\eta$, где $A\in N;\xi,\eta\in(\Sigma\cup N)^*;\alpha\in(\Sigma\cup N)^+$
}
\dfn{Неукорчивающая грамматика}{
  $\alpha\rightarrow\beta\in P\Rightarrow |\alpha|\leq|\beta|$
}
\ex{Пример КЗ-грамматики}{
  Язык $L_1=\{a^nb^nc^n:n>0\}$\\
  Грамматика, порождающая этот язык:\\
  $
  S\rightarrow aSBC\\
  S\rightarrow aBC\\
  CB\rightarrow BC\\
  aB\rightarrow ab\\
  bB\rightarrow bb\\
  bC\rightarrow bc\\
  cC\rightarrow cc\\
  $
  Она не совсем КЗ. Чтобы она стала совсем КЗ, надо, чтобы правило $CB\rightarrow BC$ было заменено последовательностью правил:\\
  $CB\rightarrow C'B\\
  C'B\rightarrow C'C\\
  C'C\rightarrow BC$
}

\section{Классические категориальные грамматики}
\dfn{Элементарные категории}{
  $\mathbb{C}$ --- элементарные категории (примитивные типы). Например, часть речи в предложении.
}
\dfn{Категория}{
  \begin{enumerate}
    \item Если $\Phi\in\mathbb{C}$, то $\Phi$ --- категория.
    \item Если $\Phi,\Psi$ --- категории, то $[\Phi/\Psi]$ и $[\Psi\backslash\Phi]$ --- категории.
  \end{enumerate}
  $Cat(\mathbb{C})$ --- множество всех категорий над $\mathbb{C}$.
}

\mprop{Правила сокращения}{
  $\Gamma_1,\Gamma_2\in Cat(\mathbb{C})^*$
  \begin{enumerate}
    \item $\Gamma_1[\Phi/\Psi]\Psi\Gamma_2\vdash\Gamma_1\Phi\Gamma_2$
    \item $\Gamma_1\Psi[\Psi\backslash\Phi]\Gamma_2\vdash\Gamma_1\Phi\Gamma_2$
  \end{enumerate}
}
\dfn{Классическая категориальная грамматика}{
  Четвёрка $G=(\Sigma,\mathbb{C},S,\lambda)$, где:\\
  $\Sigma$ --- алфавит терминалов;\\
  $\mathbb{C}$ --- множество элементарных категорий;\\
  $S\in\mathbb{C}$ --- главная категория;\\
  $\lambda:\Sigma\rightarrow P_{fin}(Cat(\mathbb{C}))$ --- словарь
  \nt{
    $w=a_1a_2\dots a_n$\\
    $\lambda(w)=\{\Phi_1,\Phi_2,\dots,\Phi_n: \Phi_i\in\lambda(a_i)\}$
  }
}
\dfn{Язык, порождаемый грамматикой $G$}{
  $L(G)=\{a_1a_2\dots a_n\in\Sigma^+:\text{ существуют $\Phi_i\in\lambda(a_i)$ такие, что }\\\Phi_1\Phi_2\dots\Phi_n\vdash^* S\}$
}

\dfn{Подкатегории}{
  $\Phi\in Cat(\mathbb{C})$\\
  $Subcat(\Phi)$ --- множество подкатегорий категории $\Phi$
  \begin{enumerate}
    \item Если $\Phi\in\mathbb{C}$, то\\
      $Subcat(\Phi)=\{\Phi\}$
    \item Если $\Phi=[\Psi/\Theta]$ или $\Phi=[\Theta\backslash\Psi]$, то\\
    $Subcat(\Phi)=Subcat(\Psi)\cup\{\Phi\}$
  \end{enumerate}

  $G=(\Sigma,\mathbb{C},S,\lambda)$\\
  $Subcat(G)$ --- множество подкатегорий категорий из $\lambda$
}

\thm{}{
  По любой классической категориальной грамматике можно построить эквивалентную КС-грамматику.
}
\pf{Доказательство}{
  $G=(\Sigma,\mathbb{C},S,\lambda)$\\
  $G'=(N,\Sigma, P,S)$\\
  $N=Subcat(G)$
  \begin{enumerate}
    \item $\Phi\in\lambda(a)\Rightarrow\Phi\rightarrow a\in P$
    \item $[\Phi/\Psi],\Psi\in N\Rightarrow\Phi\rightarrow [\Phi/\Psi]\Psi\in P$
    \item $[\Psi\backslash\Phi],\Psi\in N\Rightarrow\Psi[\Psi\backslash\Phi]\in P$
  \end{enumerate}
  \mclm{Утверждение}{
    Пусть $\Phi_1',\dots,\Phi'_n,\Psi\in N$. Тогда $\Phi_1',\dots\Phi'_n\Psi\vdash^* \Psi\Leftrightarrow\Psi\Rightarrow^*\Phi_1\dots\Phi_n$
  }
  \pf{Доказательство}{
    \textbf{Индукция по $n$}\\
    \underline{Базис}: $n=1, \Phi_1=\Psi$\\
    \underline{Индукционный шаг}:
    $(\Rightarrow)\Phi_1\dots\Phi_n\vdash^*\Psi$\\
    а)$\Phi_1\dots\Phi_i\vdash^*[\Psi/\Theta]$\\
    $\Phi_{i+1}\dots\Phi_n\vdash^*\Theta$\\
    По индукционному предположению:\\
    $[\Psi/\Theta]\Rightarrow^*\Phi_1\dots\Phi_i$;\\
    $\Theta\Rightarrow^*\Phi_{i+1}\dots\Phi_n$\\
    $\Theta\in N$\\
    $\Psi\Rightarrow[\Psi/\Theta]\Theta\Rightarrow^*\Phi_1\dots\Phi_n$

    \vspace{1em}\noindent
    б)$\Phi_1\dots\Phi_i\vdash^*\Theta$\\
    $\Phi_{i+1}\dots\Phi_n\vdash^*[\Theta\backslash\Psi]$\\
    Аналогично

    \noindent$(\Leftarrow)\Psi\Rightarrow^*\Phi_1\dots\Phi_n$\\
    а)$\Psi\rightarrow[\Psi/\Theta]\Theta$\\
    $[\Psi/\Theta]\Rightarrow^*\Phi_1\dots\Phi_i$\\
    $\Theta\Rightarrow^*\Phi_{i+1}\dots\Phi_n$\\
    По индукционному предположению:\\
    $\Phi_1\dots\Phi_i\vdash^*[\Psi/\Theta]$\\
    $\Phi_{i+1}\dots\Phi_n\vdash^*\Theta$\\
    $\Phi_1\dots\Phi_n\vdash^*[\Psi/\Theta]\Theta\vdash\Psi$

    \vspace{1em}\noindent
    б)$\Psi\rightarrow\Theta[\Theta\backslash\Psi]$\\
    Аналогично.
  }
}
